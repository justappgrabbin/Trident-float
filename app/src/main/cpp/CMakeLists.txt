cmake_minimum_required(VERSION 3.22.1)
project(trident_model LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LLAMA_SOURCE_DIR)
  set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/llama.cpp")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
set_property(TARGET ggml-base ggml-cpu ggml llama PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(trident_model SHARED model.cpp)
if(ANDROID)
  target_link_libraries(trident_model PRIVATE llama android log)
else()
  find_package(JNI REQUIRED)
  target_include_directories(trident_model PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(trident_model PRIVATE llama)
endif()
